cmake_minimum_required(VERSION 3.16)
project(pearos-about-mac LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Widgets)
qt_add_executable(pearos-about-mac src/main.cpp)
qt_add_qml_module(pearos-about-mac URI PearAboutMac VERSION 1.0 QML_FILES src/main.qml)
target_link_libraries(pearos-about-mac PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Widgets)
install(TARGETS pearos-about-mac DESTINATION bin)
install(FILES pearos-about-mac.desktop DESTINATION share/applications)
install(FILES src/Apple_logo_black.svg DESTINATION share/icons/hicolor/512x512/apps RENAME pearos-about-mac.svg)
