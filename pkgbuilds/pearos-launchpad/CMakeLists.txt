cmake_minimum_required(VERSION 3.16)
project(pearos-launchpad LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
qt_add_executable(pearos-launchpad src/main.cpp)
qt_add_qml_module(pearos-launchpad URI Launchpad VERSION 1.0 QML_FILES src/main.qml)
target_link_libraries(pearos-launchpad PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
install(TARGETS pearos-launchpad DESTINATION bin)
